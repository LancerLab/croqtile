1. tail handling

__co__ foo(int N) {
  parallel q by 6 {...}
  parallel q by N % 6 {...}
}

alternative:

__co__ foo(f32 [N] a) {
  f32 [ceil_div(N, 6) * 6] gbuffer;
  dma.pad<...> a => gbuffer;
  parallel q by 6 {...}
}
