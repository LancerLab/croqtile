auto Conv2d_kernel = [&](auto& lhs_buf, auto& rhs_buf, auto& out_buf,
                  auto ci_index, auto ci_slice_count) {
  factor::Conv2d conv2d_params(
      lhs_buf,          /* c_func_param.lhs */
      rhs_buf,          /* c_func_param.rhs */
      // out_buf,       /* fusion_param.bias */
      out_buf,          /* c_func_param.outputs */
      ci_index,         /* c_func_param.reduce_loop_index */
      ci_slice_count,   /* c_func_param.reduce_loop_cnt */
      0,                /* c_func_param.co_loop_index */
      conv_type, data_format, opt_level,
      true,             /* conv_gen_param.padding */
      {strideR, strideS}, /* strides*/
      {dilateR, dilateS}, /* rhs dilation*/
      lhs_sip_shape[0], /* conv_gen_param.N */
      lhs_sip_shape[1], /* conv_gen_param.Hi */
      lhs_sip_shape[2], /* conv_gen_param.Wi */
      out_sip_shape[1], /* conv_gen_param.Ho */
      out_sip_shape[2], /* conv_gen_param.Wo */
      ci_slice,         /* conv_gen_param.Ci */
      rhs_sip_shape[3], /* conv_gen_param.Co */
      rhs_sip_shape[0], /* conv_gen_param.r */
      rhs_sip_shape[1], /* conv_gen_param.s */
      Activation::no_activation);
  conv2d_(conv2d_params);
};

conv_dataflow(f32 mdspan<4> lhs_data, 
              f32 mdspan<4> rhs_data, 
              f32 mdspan<4> output,
              bool lhs_parallel,
              bool dma_broadcast,
              ituple lhs_padding = [0, 0, 0, 0],
              ituple strides = [1, 1],
              ituple lhs_dilation = [1, 1],
              ituple rhs_dilation = [1, 1],
              int reduce_dim,
              int csb_prefetch_depth,
              int sip_prefetch_depth,
              int feature_group_cnt) {
  // set lhs_span_h, rhs_span_h, out_span_h
  lhs_span_h : {4, 28, 28, 256};
  rhs_span_h : {3, 3, 256, 128};
  out_span_h : {4, 28, 28, 128};

  // set lhs_span_c, rhs_span_c, out_span_c
  ...
  // set lhs_span_s, rhs_span_s, out_span_s
  ...

  parallel_dims : {4, 1, 1, 1}

  // handle broadcast patterns
  bool lhs_broadcast = lhs_parallel & dma_broadcast;
  bool rhs_broadcast = !lhs_parallel & dma_broadcast;

  // QUIZ: bool+condition or enum+switch-case
  // very complex dataflow, may use bool and enum for conditioning, and then define specialised tile-sizes case-by-case
  // handle cache patterns
  bool uncache_reduce = true : rhs_span_h(reduce_dim) > rhs_span_c(reduce_dim); 
  bool cacheL2_rhs_reduce = true : !uncache_reduce && rhs_span_c(reduce_dim) > rhs_span_s(reduce_dim); 
  bool needcache_lhs_reduce = true : rhs_span_h(reduce_dim) > rhs_span_c(reduce_dim) && rhs_span_c(reduce_dim) > rhs_span_s(reduce_dim)
  bool cacheL1_rhs_reduce = !uncache_reduce && !cacheL2_rhs_reduce && needcache_lhs_reduce; 
  bool cacheL1_reduce = !uncache_reduce && !cacheL2_rhs_reduce && !cacheL1_rhs_reduce;
  pingpong = {1, 1, 1, 1, 1, 1};
  pingpong = {csb_prefetch_depth, 1, csb_prefetch_depth, sip_prefetch_depth, 1, sip_prefetch_depth} : cacheL1_reduce;
  pingpong = {csb_prefetch_depth, 1, csb_prefetch_depth, sip_prefetch_depth, 1, 1} : cacheL1_rhs_reduce;
  pingpong = {csb_prefetch_depth, 1, csb_prefetch_depth, sip_prefetch_depth, sip_prefetch_depth, 1} : cacheL2_rhs_reduce;
  pingpong = {csb_prefetch_depth, csb_prefetch_depth, csb_prefetch_depth, sip_prefetch_depth, sip_prefetch_depth, 1} : uncache_reduce;

  // handle stride, padding and dilation
  // reconfig LHS's csb and sip tile size
  // QUIZ1: how to demonstrate the inherent size dependency btw, input and output, if not by follow explicit way
  // QUIZ2: can we set tuple and span calc on indexed ranges
  // lhs_span_c(1) = (out_span_c(1) - 1) * strides(0) + dilation(0);
  // lhs_span_c(2) = (out_span_c(2) - 1) * strides(1) + dilation(1);
  lhs_span_c(1:2) = (out_span_c(1:2) - 1) * strides(0:1) + dilation(0:1);
  lhs_span_s(1:2) = (out_span_s(1:2) - 1) * strides(0:1) + dilation(0:1);

  // merge dimensions at cdma level
  // SUGGEST: support dim merge
  lhs_span_c = lhs_span_c {(0), (1), (-1), (2, 3)} : feature_group_cnt == 1 
                                                   && lhs_span_h(3) <= 64
                                                   && lhs_span_c(3) <= 64
                                                   && ...;
  out_span_c = out_span_c {(0), (1), (-1), (2, 3)} : feature_group_cnt == 1
                                                   && out_span_h(3) <= 64
                                                   && out_span_h(3) == out_span_c(3);

  lhs_c : lhs_span_h / lhs_span_c;
  rhs_c : rhs_span_h / rhs_span_c;
  out_c : out_span_h / out_span_c;


  parallel p by parallel_dims {
    if cacheL1_reduce {
      lhs_c_non_reduce = lhs_c {(0), (1), (3)};
      lhs_c_reduce = lhs_c {(2)};

      // one-by-one idx-span mapping
      with idx_lhs_c_non_reduce in lhs_c_non_reduce,
           idx_lhs_c_reduce in lhs_c_reduce,
           idx_rhs_c in rhs_c,
           idx_out_c in out_c,
           idx_group in feature_group_cnt
      where idx_lhs_c_non_reduce == idx_out_c [(0), (1), (3)] 
      and   idx_lhs_c_reduce == idx_out_c [(2)]
      and   idx_lhs_s == idx_out_s {
        foreach idx_lhs_c_non_reduce, idx_ot_c [(0), (1), (3)] {
          cdt_input = dte.linear input.chunkAt(p, idx_in_c) -> shared;
          foreach idx_in_c_reduce {
            cdt_input = dte.linear input.chunkAt(p, idx_in_c) -> shared;
            foreach idx_lhs_c(reduce_dim) {
              foreach idx_in_s, idx_ot_s {
                sdt_input = dte.pad<[1, 1], [1, 1], [0, 0], 0> cdt_input.chunkAt(p, idx_in_s) -> local;
                wait sdt_input
                local output = Conv2d_kernel(sdt_input.lhs, sdt_input.rhs, reduce_dim, lhs_span_c(3) / rhs_span_s(3))
                sdt_output = dte.pad<[1, 1], [1, 1], [0, 0], 0> sdt_input -> output.chunkAt(p, idx_ot_s);
              }
              cdt_output = dte.linear sdt_output -> output.chunkAt(p, idx_ot_c);
            }
          }
        }
      }
    } else if cacheL2_rhs_reduce {
      rhs_c_non_reduce = rhs_c {(0), (1), (3)};
      rhs_c_reduce = rhs_c {(2)};

      // one-line idx-span mapping
      with idx_rhs_c_non_reduce, idx_rhs_c_reduce, idx_lhs_c, idx_out_c in rhs_c_non_reduce, rhs_c_reduce, lhs_c, out_c,
           idx_group in feature_group_cnt
      where idx_rhs_c_non_reduce == idx_out_c [(0), (1), (3)] 
      and   idx_rhs_c_reduce == idx_out_c [(2)]
      and   idx_rhs_s == idx_out_s {
        foreach idx_rhs_c_non_reduce, idx_ot_c [(0), (1), (3)] {
          cdt_input = dte.linear input.chunkAt(p, idx_in_c) -> shared;
          foreach idx_in_c_reduce {
            cdt_input = dte.linear input.chunkAt(p, idx_in_c) -> shared;
            foreach idx_rhs_c(reduce_dim) {
              foreach idx_in_s, idx_ot_s {
                sdt_input = dte.pad<[1, 1], [1, 1], [0, 0], 0> cdt_input.chunkAt(p, idx_in_s) -> local;
                wait sdt_input
                local output = Conv2d_kernel(sdt_input.rhs, sdt_input.lhs, reduce_dim, rhs_span_c(3) / lhs_span_s(3))
                sdt_output = dte.pad<[1, 1], [1, 1], [0, 0], 0> sdt_input -> output.chunkAt(p, idx_ot_s);
              }
              cdt_output = dte.linear sdt_output -> output.chunkAt(p, idx_ot_c);
            }
          }
        }
      }
    } else if
     ...
  }
}
