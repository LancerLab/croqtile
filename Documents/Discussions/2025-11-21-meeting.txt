Discussion:
1. how to hint the invalid code:

   parallel {m, n} by [M / MMA_M / 4, N / MMA_N / 4] : block {
     mc = mma.fill 0.0;  // c = {0}
     parallel g0 by 4: group {
       foreach k in K / MMA_K {
         ma = mma.load lhs.chunkat(m#g0, k);
         mb = mma.load rhs.chunkat(k, n#g0);
         mma.row.col mc, ma, mb;   // row * col : c += a * b;
       }
       mma.store mc, output.chunkat(m#g0, n#g0);
     }
   }

    parallel {m, n} by [M / MMA_M, N / MMA_N] : block {
      mc = mma.fill 0.0;  // c = {0}
      parallel g by 4: group {
        foreach k in K / MMA_K {
          ma = mma.load lhs.chunkat(m, k#g);
          mb = mma.load rhs.chunkat(k#g, n);
          mma.row.col mc, ma, mb;   // row * col : c += a * b;
        }
        mma.store mc, output.chunkat(m, n);
      }
    }
