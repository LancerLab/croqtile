__co__ void sgemm(f32 [4096, 4096] lhs, f32 [4096, 4096] rhs, f32 [4096, 4096] output) { /// device program
  parallel (p, q) by [128, 128] {  // p is sip_index
    with index = {m_tile, n_tile, k_tile} in [32, 32, 1] {  // decl your tiled spans
      shared f32[1, 1] l2_out; // alternative: f32[lhs.span(0)/#p, lhs.span(1)/#x, rhs.span(2)/#y]
      foreach k_tile {
        lhs_load = dma.copy lhs.chunkat(p*m_tile, k_tile) => shared; // tiling_factor: {6, 17, 4}, new_size {lhs(0)/6, lhs(1)/17, lhs(2)/4}
        rhs_load = dma.copy rhs.chunkat(k_tile, q*n_tile) => shared; // no explicit decl the target buffer, but can refer by <FutureType>.data
        call sgemm_kernel(lhs_load.data, rhs_load.data, l2_out); // |span| calculate the total size of span.
        out_store = dma.copy l2_out => output.chunkat(p*m_tile, q*n_tile); // can use defined vars
      }
    }
  } // can use python styling
}
